cmake_minimum_required(VERSION 3.16)
project(M68kStepTests C)
enable_testing()
add_executable(m68k_data_region_test data_region_test.c ../../runtime/m68k_step.c)
target_include_directories(m68k_data_region_test PRIVATE ../../runtime)
